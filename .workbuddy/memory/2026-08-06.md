interface RawBatchX { /* 后端原始字段名 + 嵌套结构 */ }

const normalizeBatchX = (raw?: RawBatchX | null): BatchXData => {
  const summary = raw?.summary ?? {};
  const list = Array.isArray(raw?.predictions) ? raw!.predictions! : [];
  return {
    summary: {
      a: typeof summary.a === 'number' ? summary.a : 0,
      b: summary.b ?? 0,
    },
    predictions: list.map(item => ({
      // 关键字段提顶层 / 重命名 / 兜底
      name: item?.name ?? '未知',
      score: typeof item?.nested?.score === 'number' ? item.nested.score : 0,
    })),
  };
};

// 调用：async await
getX: async (params) => {
  const raw = (await request(`/api/algorithm/x?${qs}`)) as RawBatchX | null;
  return normalizeBatchX(raw);
},
