- **位置**: `backend/app.py` (line 438)
- **现象**: `初始化RBAC数据失败: No module named 'routes.rbac_routes'`
- **根因**: 模块路径错误，应为 `api.users.rbac_routes`
- **影响**: 系统缺少默认权限和角色数据
- **状态**: 待修复

#### 问题3：日志文件滚动权限错误
- **位置**: `backend/utils/structured_logger.py`
- **现象**: `PermissionError: [WinError 32] 另一个程序正在使用此文件`
- **根因**: Windows下文件锁冲突，标准 `RotatingFileHandler` 不支持多进程
- **影响**: 日志丢失，日志文件无法自动归档
- **状态**: 待修复

#### 问题4：数据库初始化异步竞态
- **位置**: `backend/app.py` (line 441)
- **现象**: 数据库初始化在后台线程执行，首次请求可能失败
- **根因**: `db_init_thread` 为 daemon 线程，主线程不等待完成
- **影响**: 首次API请求可能收到数据库错误
- **状态**: 待修复

#### 问题5：MQTT双连接资源浪费
- **位置**: `backend/app.py` (line 465+)
- **现象**: 同时维护TCP(8883)和WebSocket(8084)两种MQTT连接
- **根因**: `init_mqtt()` 同时创建两个连接实例
- **影响**: 资源浪费，连接管理复杂
- **状态**: 待优化

#### 问题6：前端状态重复存储

- **现象**: `WebSocketStore` 和 `DeviceStore` 都存储 `deviceStatuses`

from functools import wraps
from flask import request, jsonify

class RequestValidator:
    @staticmethod
    def validate_json(schema):
        def decorator(f):
            @wraps(f)
            def wrapper(*args, **kwargs):
                if not request.is_json:
                    return jsonify({'error': 'Content-Type必须是application/json'}), 400
                data = request.get_json()
                errors = schema.validate(data)
                if errors:
                    return jsonify({'error': '数据验证失败', 'details': errors}), 422
                return f(*args, **kwargs)
            return wrapper
        return decorator

#### 4.1.4 优化AnimatedNumber组件

**当前问题**: 每次数值变化都触发动画，CPU占用高

**优化方案**:
```typescript
const AnimatedNumber = memo(({ value, className = '' }: Props) => {
  const [displayValue, setDisplayValue] = useState(value);
  const rafRef = useRef<number>();
  
  useEffect(() => {
    // 防抖：快速变化时只动画最后一次
    const timeoutId = setTimeout(() => {
      const startTime = Date.now();
      const startValue = displayValue;
      const duration = 300;
      
      const animate = () => {
        const elapsed = Date.now() - startTime;
        const progress = Math.min(elapsed / duration, 1);
        
        // 使用 easeOutQuart 缓动函数
        const eased = 1 - Math.pow(1 - progress, 4);
        const current = startValue + (value - startValue) * eased;
        
        setDisplayValue(Math.floor(current));
        
        if (progress < 1) {
          rafRef.current = requestAnimationFrame(animate);
        }
      };
      
      rafRef.current = requestAnimationFrame(animate);
    }, 50); // 50ms防抖
    
    return () => {
      clearTimeout(timeoutId);
      if (rafRef.current) cancelAnimationFrame(rafRef.current);
    };
  }, [value]);
  
  return <span className={className}>{displayValue.toLocaleString()}</span>;
}, (prev, next) => prev.value === next.value);
```

#### 4.1.5 优化WebSocket重连逻辑

**当前代码**: 简单重连，无限重试

**优化方案**:
```typescript
// 指数退避重连策略
const RECONNECT_CONFIG = {
  initialDelay: 1000,
  maxDelay: 30000,
  maxAttempts: 10,
  backoffMultiplier: 2,
};

function useWebSocketWithBackoff(url: string) {
  const [isConnected, setIsConnected] = useState(false);
  const attemptRef = useRef(0);
  const timeoutRef = useRef<NodeJS.Timeout>();
  
  const connect = useCallback(() => {
    const socket = io(url, {
      transports: ['websocket', 'polling'],
      reconnection: false, // 手动控制重连
    });
    
    socket.on('connect', () => {
      attemptRef.current = 0;
      setIsConnected(true);
    });
    
    socket.on('disconnect', () => {
      setIsConnected(false);
      scheduleReconnect();
    });
    
    return socket;
  }, [url]);
  
  const scheduleReconnect = useCallback(() => {
    if (attemptRef.current >= RECONNECT_CONFIG.maxAttempts) {
      console.error('WebSocket重连次数已达上限');
      return;
    }
    
    const delay = Math.min(
      RECONNECT_CONFIG.initialDelay * Math.pow(RECONNECT_CONFIG.backoffMultiplier, attemptRef.current),
      RECONNECT_CONFIG.maxDelay
    );
    
    timeoutRef.current = setTimeout(() => {
      attemptRef.current++;
      connect();
    }, delay);
  }, [connect]);
  
  return { isConnected, connect };
}
```

#### 4.2.2 API响应缓存

**优化方案**:
```python
# backend/utils/cache_middleware.py
from functools import wraps
from flask import request, jsonify
import json
import hashlib

class ApiCache:
    def __init__(self, redis_client, default_ttl=300):
        self.redis = redis_client
        self.default_ttl = default_ttl
    
    def cache_key(self, prefix='api'):
        """生成缓存key"""
        data = f"{request.method}:{request.path}:{json.dumps(request.args)}"
        return f"{prefix}:{hashlib.md5(data.encode()).hexdigest()}"
    
    def cached(self, ttl=None, key_prefix='api'):
        def decorator(f):
            @wraps(f)
            def wrapper(*args, **kwargs):
                # POST/PUT/DELETE 不缓存
                if request.method != 'GET':
                    return f(*args, **kwargs)
                
                cache_key = self.cache_key(key_prefix)
                
                # 尝试读取缓存
                cached = self.redis.get(cache_key)
                if cached:
                    return jsonify(json.loads(cached))
                
                # 执行原函数
                response = f(*args, **kwargs)
                
                # 缓存响应
                if hasattr(response, 'get_json'):
                    self.redis.setex(
                        cache_key,
                        ttl or self.default_ttl,
                        json.dumps(response.get_json())
                    )
                
                return response
            return wrapper
        return decorator

# 使用示例
api_cache = ApiCache(redis_client)

@api.route('/users')
@api_cache.cached(ttl=60, key_prefix='users')
def get_users():
    # 查询逻辑...
    pass
```

#### 4.2.3 MQTT消息并行处理

**当前问题**: 单线程处理MQTT消息，高峰期堆积

**优化方案**:
```python
# backend/services/mqtt_processor.py
from concurrent.futures import ThreadPoolExecutor
import queue

class MQTTMessageProcessor:
    def __init__(self, max_workers=4):
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
        self.topic_handlers = {}
        self.priority_queue = queue.PriorityQueue()
        
    def register_handler(self, topic_pattern, handler, priority=5):
        """注册主题处理器"""
        self.topic_handlers[topic_pattern] = {
            'handler': handler,
            'priority': priority
        }
    
    def process_message(self, topic, message):
        """提交消息到线程池处理"""
        handler_info = self.topic_handlers.get(topic)
        if not handler_info:
            return
        
        # 高优先级直接处理
        if handler_info['priority'] <= 2:
            handler_info['handler'](topic, message)
        else:
            # 普通优先级提交线程池
            self.executor.submit(handler_info['handler'], topic, message)
    
    def shutdown(self):
        self.executor.shutdown(wait=True)
```

# 2. 批量写入优化
class BatchWriter:
    def __init__(self, db, batch_size=100, flush_interval=1.0):
        self.db = db
        self.batch_size = batch_size
        self.flush_interval = flush_interval
        self.buffer = []
        self.last_flush = time.time()
    
    def add(self, obj):
        self.buffer.append(obj)
        
        if (len(self.buffer) >= self.batch_size or 
            time.time() - self.last_flush >= self.flush_interval):
            self.flush()
    
    def flush(self):
        if not self.buffer:
            return
        
        try:
            self.db.session.add_all(self.buffer)
            self.db.session.commit()
            self.buffer = []
            self.last_flush = time.time()
        except Exception as e:
            self.db.session.rollback()
            raise

import os

class SecurityConfig:
    # 强制生产环境检查
    @classmethod
    def validate_production(cls):
        if os.getenv('FLASK_ENV') == 'production':
            assert os.getenv('FLASK_SECRET_KEY'), \
                '生产环境必须设置 FLASK_SECRET_KEY'
            assert os.getenv('DATABASE_URI') != 'sqlite:///...', \
                '生产环境不建议使用SQLite'
    
    # 统一权限验证装饰器
    @staticmethod
    def require_permission(permission_code):
        def decorator(f):
            @wraps(f)
            def wrapper(*args, **kwargs):
                from flask import g
                if not g.current_user or not g.current_user.has_permission(permission_code):
                    return jsonify({'error': '权限不足'}), 403
                return f(*args, **kwargs)
            return wrapper
        return decorator
