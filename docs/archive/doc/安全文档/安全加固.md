```python
# utils/security.py 增强

import jwt
import hashlib
from datetime import datetime, timedelta
from functools import wraps

class TokenManager:
    """JWT令牌管理器"""
    
    # 令牌有效期配置
    ACCESS_TOKEN_EXPIRES = timedelta(hours=24)
    REFRESH_TOKEN_EXPIRES = timedelta(days=7)
    
    # 令牌黑名单（用于撤销）
    _token_blacklist = set()
    
    @classmethod
    def create_access_token(cls, user_id, role='admin'):
        """创建访问令牌"""
        payload = {
            'user_id': user_id,
            'role': role,
            'type': 'access',
            'iat': datetime.utcnow(),
            'exp': datetime.utcnow() + cls.ACCESS_TOKEN_EXPIRES,
            'jti': hashlib.sha256(str(datetime.utcnow()).encode()).hexdigest()
        }
        return jwt.encode(payload, Config.JWT_SECRET_KEY, algorithm='HS256')
    
    @classmethod
    def create_refresh_token(cls, user_id):
        """创建刷新令牌"""
        payload = {
            'user_id': user_id,
            'type': 'refresh',
            'iat': datetime.utcnow(),
            'exp': datetime.utcnow() + cls.REFRESH_TOKEN_EXPIRES,
            'jti': hashlib.sha256(str(datetime.utcnow()).encode()).hexdigest()
        }
        return jwt.encode(payload, Config.JWT_SECRET_KEY, algorithm='HS256')
    
    @classmethod
    def verify_token(cls, token, token_type='access'):
        """验证令牌"""
        try:
            # 检查是否在黑名单
            payload = jwt.decode(token, Config.JWT_SECRET_KEY, algorithms=['HS256'])
            if payload.get('jti') in cls._token_blacklist:
                return None, 'Token has been revoked'
            
            if payload.get('type') != token_type:
                return None, 'Invalid token type'
            
            return payload, None
            
        except jwt.ExpiredSignatureError:
            return None, 'Token has expired'
        except jwt.InvalidTokenError:
            return None, 'Invalid token'
    
    @classmethod
    def revoke_token(cls, token):
        """撤销令牌"""
        try:
            payload = jwt.decode(token, Config.JWT_SECRET_KEY, algorithms=['HS256'], options={'verify_exp': False})
            cls._token_blacklist.add(payload['jti'])
            return True
        except:
            return False
```

```python
# utils/password_policy.py

import re
from datetime import datetime, timedelta

class PasswordPolicy:
    """密码策略"""
    
    MIN_LENGTH = 8
    MAX_LENGTH = 128
    REQUIRE_UPPERCASE = True
    REQUIRE_LOWERCASE = True
    REQUIRE_DIGIT = True
    REQUIRE_SPECIAL = True
    SPECIAL_CHARS = '!@#$%^&*()_+-=[]{}|;:,.<>?'
    
    # 密码历史（不能重复使用最近N次密码）
    PASSWORD_HISTORY_SIZE = 5
    
    # 密码强度正则
    STRENGTH_PATTERNS = {
        'weak': r'^[a-zA-Z]+$|^[0-9]+$',
        'medium': r'^(?=.*[a-z])(?=.*[A-Z])|(?=.*[a-z])(?=.*[0-9])',
        'strong': r'^(?=.*[a-z])(?=.*[A-Z])(?=.*[0-9])(?=.*[!@#$%^&*])'
    }
    
    @classmethod
    def validate(cls, password, old_passwords=None):
        """验证密码强度
        
        Args:
            password: 新密码
            old_passwords: 历史密码列表
        
        Returns:
            tuple: (是否有效, 错误信息或密码强度)
        """
        errors = []
        
        # 检查长度
        if len(password) < cls.MIN_LENGTH:
            errors.append(f'密码长度至少{cls.MIN_LENGTH}位')
        if len(password) > cls.MAX_LENGTH:
            errors.append(f'密码长度最多{cls.MAX_LENGTH}位')
        
        # 检查复杂度
        if cls.REQUIRE_UPPERCASE and not re.search(r'[A-Z]', password):
            errors.append('必须包含大写字母')
        
        if cls.REQUIRE_LOWERCASE and not re.search(r'[a-z]', password):
            errors.append('必须包含小写字母')
        
        if cls.REQUIRE_DIGIT and not re.search(r'[0-9]', password):
            errors.append('必须包含数字')
        
        if cls.REQUIRE_SPECIAL:
            has_special = any(c in cls.SPECIAL_CHARS for c in password)
            if not has_special:
                errors.append(f'必须包含特殊字符: {cls.SPECIAL_CHARS}')
        
        if errors:
            return False, errors
        
        # 检查密码历史
        if old_passwords:
            for old_pwd in old_passwords[-cls.PASSWORD_HISTORY_SIZE:]:
                if cls._hash_password(password) == old_pwd:
                    return False, ['不能使用最近使用过的密码']
        
        # 计算密码强度
        strength = cls._calculate_strength(password)
        return True, strength
    
    @classmethod
    def _calculate_strength(cls, password):
        """计算密码强度"""
        score = 0
        
        # 长度分数
        if len(password) >= 12:
            score += 2
        elif len(password) >= 8:
            score += 1
        
        # 字符类型分数
        if re.search(r'[a-z]', password):
            score += 1
        if re.search(r'[A-Z]', password):
            score += 1
        if re.search(r'[0-9]', password):
            score += 1
        if any(c in cls.SPECIAL_CHARS for c in password):
            score += 1
        
        # 转换为强度等级
        if score >= 5:
            return 'strong'
        elif score >= 3:
            return 'medium'
        return 'weak'
    
    @classmethod
    def _hash_password(cls, password):
        """哈希密码"""
        import hashlib
        return hashlib.sha256(password.encode()).hexdigest()
```

```python
# utils/input_validator.py

import re
from html import escape

class InputValidator:
    """输入验证器"""
    
    # 白名单验证
    ALLOWED_CARD_ID_PATTERN = r'^[A-Z0-9]{8,16}$'
    ALLOWED_USERNAME_PATTERN = r'^[a-zA-Z0-9_]{3,20}$'
    
    @classmethod
    def validate_card_id(cls, card_id):
        """验证卡号"""
        if not card_id:
            return False, '卡号不能为空'
        
        if not re.match(cls.ALLOWED_CARD_ID_PATTERN, card_id):
            return False, '卡号格式不正确'
        
        return True, None
    
    @classmethod
    def validate_username(cls, username):
        """验证用户名"""
        if not username:
            return False, '用户名不能为空'
        
        if not re.match(cls.ALLOWED_USERNAME_PATTERN, username):
            return False, '用户名格式不正确（3-20位字母数字下划线）'
        
        return True, None
    
    @classmethod
    def sanitize_html(cls, text):
        """HTML转义"""
        if not text:
            return ''
        return escape(text)
    
    @classmethod
    def check_sql_injection(cls, text):
        """检测SQL注入"""
        sql_patterns = [
            r"(\bunion\b.*\bselect\b)",
            r"(\bselect\b.*\bfrom\b)",
            r"(\binsert\b.*\binto\b)",
            r"(\bdelete\b.*\bfrom\b)",
            r"(\bdrop\b.*\btable\b)",
            r"(--|;|'|\"|\\*)"
        ]
        
        text_lower = text.lower()
        for pattern in sql_patterns:
            if re.search(pattern, text_lower, re.IGNORECASE):
                return False, '检测到非法字符'
        
        return True, None
    
    @classmethod
    def check_xss(cls, text):
        """检测XSS攻击"""
        xss_patterns = [
            r'<script',
            r'javascript:',
            r'onerror=',
            r'onload=',
            r'<iframe',
            r'<object',
            r'<embed'
        ]
        
        text_lower = text.lower()
        for pattern in xss_patterns:
            if re.search(pattern, text_lower):
                return False, '检测到潜在攻击代码'
        
        return True, None
```

```python
# utils/param_validator.py

from functools import wraps
from flask import request, jsonify

def validate_params(*required_fields, **field_types):
    """参数验证装饰器
    
    用法:
    @validate_params('username', 'password', email=str, age=int)
    def login():
        pass
    """
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            data = request.get_json() or {}
            
            # 检查必需字段
            for field in required_fields:
                if field not in data:
                    return jsonify({
                        'success': False,
                        'error': f'缺少必需参数: {field}'
                    }), 400
            
            # 验证字段类型
            for field, expected_type in field_types.items():
                if field in data:
                    value = data[field]
                    if not isinstance(value, expected_type):
                        return jsonify({
                            'success': False,
                            'error': f'{field} 类型错误，期望 {expected_type.__name__}'
                        }), 400
            
            return func(*args, **kwargs)
        return wrapper
    return decorator


def validate_score_range(score, min_val=0, max_val=1000):
    """验证积分范围"""
    if not isinstance(score, (int, float)):
        return False, '积分必须是数字'
    if score < min_val or score > max_val:
        return False, f'积分必须在{min_val}到{max_val}之间'
    return True, None


def validate_pagination(page, per_page, max_per_page=100):
    """验证分页参数"""
    try:
        page = int(page)
        per_page = int(per_page)
    except (TypeError, ValueError):
        return False, '分页参数必须是整数'
    
    if page < 1:
        return False, '页码必须大于0'
    if per_page < 1 or per_page > max_per_page:
        return False, f'每页数量必须在1到{max_per_page}之间'
    
    return True, None
```

```python
from cryptography.fernet import Fernet
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2
from cryptography.hazmat.backends import default_backend
import base64
import os

class DataEncryption:
    """数据加密工具"""
    
    def __init__(self, key=None):
        if key:
            self.key = base64.urlsafe_b64decode(key)
        else:
            # 从环境变量或配置文件生成
            self.key = self._generate_key()
        
        self.cipher = Fernet(self.key)
    
    def _generate_key(self):
        """生成加密密钥"""
        # 这里应该从安全的密钥管理服务获取
        from config import Config
        password = Config.SECRET_KEY.encode()
        salt = b'salt_for_key_derivation'  # 生产环境应该随机生成
        
        kdf = PBKDF2(
            algorithm=hashes.SHA256(),
            length=32,
            salt=salt,
            iterations=100000,
            backend=default_backend()
        )
        key = base64.urlsafe_b64encode(kdf.derive(password))
        return key
    
    def encrypt(self, data):
        """加密数据"""
        if isinstance(data, str):
            data = data.encode()
        return self.cipher.encrypt(data).decode()
    
    def decrypt(self, encrypted_data):
        """解密数据"""
        return self.cipher.decrypt(encrypted_data.encode()).decode()
    
    def hash_sensitive_field(self, field):
        """对敏感字段进行哈希"""
        import hashlib
        return hashlib.sha256(field.encode()).hexdigest()
```

```python
# utils/data_masking.py

import re

class DataMasking:
    """数据脱敏工具"""
    
    @staticmethod
    def mask_card_id(card_id):
        """脱敏卡号"""
        if not card_id or len(card_id) < 8:
            return '****'
        return card_id[:4] + '****' + card_id[-4:]
    
    @staticmethod
    def mask_phone(phone):
        """脱敏手机号"""
        if not phone or len(phone) < 7:
            return '****'
        return phone[:3] + '****' + phone[-4:]
    
    @staticmethod
    def mask_email(email):
        """脱敏邮箱"""
        if not email or '@' not in email:
            return '****@****'
        parts = email.split('@')
        username = parts[0]
        domain = parts[1] if len(parts) > 1 else ''
        
        if len(username) <= 2:
            masked_username = username[0] + '*'
        else:
            masked_username = username[0] + '*' * (len(username) - 2) + username[-1]
        
        return f"{masked_username}@{domain}"
    
    @staticmethod
    def mask_name(name):
        """脱敏姓名"""
        if not name or len(name) < 2:
            return '*'
        return name[0] + '*' * (len(name) - 1)
    
    @staticmethod
    def mask_ip(ip):
        """脱敏IP地址"""
        if not ip:
            return '****'
        parts = ip.split('.')
        if len(parts) == 4:
            return f"{parts[0]}.{parts[1]}.***.{parts[3]}"
        return ip
    
    @classmethod
    def mask_dict(cls, data, fields):
        """批量脱敏字典数据
        
        Args:
            data: 原始数据字典
            fields: 需要脱敏的字段及方法
        
        Returns:
            dict: 脱敏后的数据
        """
        masked = data.copy()
        for field, method in fields.items():
            if field in masked and masked[field]:
                if hasattr(cls, f'mask_{method}'):
                    masked[field] = getattr(cls, f'mask_{method}')(masked[field])
        return masked
```

```python
# utils/anomaly_detection.py

from datetime import datetime, timedelta
from collections import defaultdict
import time

class AnomalyDetector:
    """异常行为检测器"""
    
    def __init__(self):
        # 记录用户行为
        self.user_actions = defaultdict(list)
        # 记录IP访问
        self.ip_access = defaultdict(list)
        # 记录异常标记
        self.suspicious_users = set()
        self.suspicious_ips = set()
    
    def record_action(self, user_id, action, ip_address, timestamp=None):
        """记录用户行为"""
        if timestamp is None:
            timestamp = datetime.now()
        
        # 记录到用户行为
        self.user_actions[user_id].append({
            'action': action,
            'ip': ip_address,
            'timestamp': timestamp
        })
        
        # 记录到IP访问
        self.ip_access[ip_address].append({
            'user': user_id,
            'action': action,
            'timestamp': timestamp
        })
        
        # 清理过期数据（保留最近1小时）
        self._cleanup_old_data()
    
    def check_rate_anomaly(self, identifier, threshold=60, window=60):
        """检测频率异常
        
        Args:
            identifier: 用户ID或IP
            threshold: 阈值（次）
            window: 时间窗口（秒）
        
        Returns:
            bool: 是否异常
        """
        now = datetime.now()
        cutoff = now - timedelta(seconds=window)
        
        if identifier in self.user_actions:
            recent = [a for a in self.user_actions[identifier] 
                     if a['timestamp'] > cutoff]
            if len(recent) > threshold:
                return True
        
        return False
    
    def check_geo_anomaly(self, user_id, ip_address):
        """检测地理位置异常"""
        if user_id not in self.user_actions:
            return False
        
        # 获取最近操作
        recent = self.user_actions[user_id][-5:]
        
        # 检查是否有来自不同地区的访问
        if len(recent) >= 2:
            ips = set(a['ip'] for a in recent)
            if len(ips) > 2:  # 超过2个不同IP
                return True
        
        return False
    
    def check_time_anomaly(self, user_id):
        """检测时间异常（非常规时间访问）"""
        now = datetime.now()
        hour = now.hour
        
        # 定义非常规时间（凌晨2点到5点）
        if hour >= 2 and hour <= 5:
            # 记录警告
            return True
        
        return False
    
    def _cleanup_old_data(self):
        """清理过期数据"""
        now = datetime.now()
        cutoff = now - timedelta(hours=24)
        
        # 清理用户行为
        for user_id in list(self.user_actions.keys()):
            self.user_actions[user_id] = [
                a for a in self.user_actions[user_id]
                if a['timestamp'] > cutoff
            ]
            if not self.user_actions[user_id]:
                del self.user_actions[user_id]
        
        # 清理IP访问
        for ip in list(self.ip_access.keys()):
            self.ip_access[ip] = [
                a for a in self.ip_access[ip]
                if a['timestamp'] > cutoff
            ]
            if not self.ip_access[ip]:
                del self.ip_access[ip]
```

## 8. 常见安全威胁及防护

| 威胁类型 | 描述 | 防护措施 |
|---------|------|----------|
| SQL注入 | 通过输入注入恶意SQL | 使用ORM、参数化查询 |
| XSS | 跨站脚本攻击 | 输入验证、HTML转义 |
| CSRF | 跨站请求伪造 | CSRF令牌、SameSite Cookie |
| 会话劫持 | 窃取用户会话 | HTTPS、安全Cookie设置 |
| 暴力破解 | 频繁尝试登录 | 限流、账户锁定 |
| 敏感数据泄露 | 未加密存储 | 数据加密、脱敏处理 |
| 权限绕过 | 未授权访问资源 | 权限检查、角色验证 |
| 文件上传 | 上传恶意文件 | 文件类型检查、内容验证 |
