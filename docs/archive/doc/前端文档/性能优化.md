## 1. 组件性能优化

/*
1.1 使用 React.memo 优化纯展示组件
----------------------------------------*/

// ❌ 不好：每次父组件渲染，子组件都会重新渲染
const UserCard = ({ name, score }) => {
  return (
    <div>
      <span>{name}</span>
      <span>{score}</span>
    </div>
  );
};

// ✅ 好：使用 React.memo 避免不必要的重新渲染
const UserCard = memo(({ name, score }) => {
  return (
    <div>
      <span>{name}</span>
      <span>{score}</span>
    </div>
  );
});

// ✅ 更好：自定义比较函数
const UserCard = memo(
  ({ name, score }) => {
    return (
      <div>
        <span>{name}</span>
        <span>{score}</span>
      </div>
    );
  },
  (prevProps, nextProps) => {
    // 只有当 name 或 score 改变时才重新渲染
    return prevProps.name === nextProps.name && prevProps.score === nextProps.score;
  }
);

/*
1.2 使用 useMemo 缓存计算结果
----------------------------------------*/

// ❌ 不好：每次渲染都重新计算
const filteredUsers = users.filter(u => u.class_name === selectedClass);

// ✅ 好：使用 useMemo 缓存计算结果
const filteredUsers = useMemo(() => {
  return users.filter(u => u.class_name === selectedClass);
}, [users, selectedClass]);

// ✅ 用于复杂计算
const sortedData = useMemo(() => {
  return [...data].sort((a, b) => {
    // 复杂的排序逻辑
    return b.score - a.score;
  });
}, [data]);

/*
1.3 使用 useCallback 缓存回调函数
----------------------------------------*/

// ❌ 不好：每次渲染都创建新函数
const handleClick = () => {
  console.log('clicked');
};

// ✅ 好：使用 useCallback 缓存函数
const handleClick = useCallback(() => {
  console.log('clicked');
}, []);

// ✅ 依赖项正确处理
const handleUpdate = useCallback((id) => {
  setItems(items.map(item => 
    item.id === id ? { ...item, updated: true } : item
  ));
}, [items]);

3.1 合理拆分状态
----------------------------------------*/

/*
3.2 使用useReducer管理复杂状态
----------------------------------------*/

const initialState = {
  users: [],
  loading: false,
  error: null,
  filter: ''
};

function reducer(state, action) {
  switch (action.type) {
    case 'FETCH_START':
      return { ...state, loading: true, error: null };
    case 'FETCH_SUCCESS':
      return { ...state, loading: false, users: action.payload };
    case 'FETCH_ERROR':
      return { ...state, loading: false, error: action.payload };
    case 'SET_FILTER':
      return { ...state, filter: action.payload };
    default:
      return state;
  }
}

## 4. 数据获取优化
*/

/*
4.1 避免不必要的请求
----------------------------------------*/

// ✅ 使用防抖处理搜索输入
import { useDebouncedCallback } from 'use-debounce';

const SearchInput = ({ onSearch }) => {
  const debouncedSearch = useDebouncedCallback(
    (value) => {
      onSearch(value);
    },
    300
  );

  return <input onChange={(e) => debouncedSearch(e.target.value)} />;
};

// ✅ 使用节流处理频繁事件
import { useThrottledCallback } from 'use-debounce';

const ScrollHandler = ({ onScroll }) => {
  const throttledScroll = useThrottledCallback(
    (e) => {
      onScroll(e);
    },
    100
  );

  return <div onScroll={throttledScroll}>...</div>;
};
