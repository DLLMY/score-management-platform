# 配置管理指南

## 概述

本项目使用集中化配置管理，所有配置通过环境变量和配置文件统一管理，便于部署和环境切换。

## 配置文件结构

```
backend/
├── .env                    # 环境变量（不提交到版本控制）
├── .env.example            # 环境变量示例（提交到版本控制）
├── config.py              # 配置管理模块
└── instance/
    └── config.py          # 实例配置（数据库等）
```

## 配置管理模块

### config.py

```python
"""
集中化配置管理
支持多环境配置继承
"""

import os
from dotenv import load_dotenv

# 加载环境变量
load_dotenv()


class Config:
    """基础配置"""
    
    # Flask配置
    SECRET_KEY = os.getenv('SECRET_KEY', 'dev-secret-key')
    DEBUG = os.getenv('DEBUG', 'False').lower() == 'true'
    TESTING = False
    
    # 数据库配置
    SQLALCHEMY_DATABASE_URI = os.getenv(
        'DATABASE_URL',
        f"sqlite:///{os.path.join(os.path.dirname(__file__), 'instance', 'score_management.db')}"
    )
    SQLALCHEMY_TRACK_MODIFICATIONS = False
    SQLALCHEMY_ENGINE_OPTIONS = {
        'pool_size': 10,
        'pool_recycle': 3600,
        'pool_pre_ping': True
    }
    
    # Redis配置
    REDIS_HOST = os.getenv('REDIS_HOST', 'localhost')
    REDIS_PORT = int(os.getenv('REDIS_PORT', 6379))
    REDIS_PASSWORD = os.getenv('REDIS_PASSWORD', None)
    REDIS_DB = int(os.getenv('REDIS_DB', 0))
    
    # MQTT配置
    MQTT_BROKER = os.getenv('MQTT_BROKER', 'localhost')
    MQTT_PORT = int(os.getenv('MQTT_PORT', 8883))
    MQTT_USERNAME = os.getenv('MQTT_USERNAME', '')
    MQTT_PASSWORD = os.getenv('MQTT_PASSWORD', '')
    MQTT_SSL = os.getenv('MQTT_SSL', 'True').lower() == 'true'
    MQTT_KEEPALIVE = int(os.getenv('MQTT_KEEPALIVE', 60))
    
    # JWT配置
    JWT_SECRET_KEY = os.getenv('JWT_SECRET_KEY', SECRET_KEY)
    JWT_ACCESS_TOKEN_EXPIRES = int(os.getenv('JWT_ACCESS_TOKEN_EXPIRES', 86400))
    JWT_REFRESH_TOKEN_EXPIRES = int(os.getenv('JWT_REFRESH_TOKEN_EXPIRES', 604800))
    
    # CORS配置
    CORS_ORIGINS = os.getenv('CORS_ORIGINS', '*').split(',')
    CORS_SUPPORTS_CREDENTIALS = True
    
    # 安全配置
    SESSION_COOKIE_SECURE = os.getenv('SESSION_COOKIE_SECURE', 'True').lower() == 'true'
    SESSION_COOKIE_HTTPONLY = True
    SESSION_COOKIE_SAMESITE = os.getenv('SESSION_COOKIE_SAMESITE', 'Lax')
    
    # 限流配置
    RATELIMIT_STORAGE_URL = f"redis://{REDIS_HOST}:{REDIS_PORT}/1"
    RATELIMIT_DEFAULT = "1000 per hour"
    RATELIMIT_HEADERS_ENABLED = True
    
    # 日志配置
    LOG_LEVEL = os.getenv('LOG_LEVEL', 'INFO')
    LOG_DIR = os.getenv('LOG_DIR', 'logs')
    
    # 性能配置
    MAX_CONTENT_LENGTH = int(os.getenv('MAX_CONTENT_LENGTH', 16 * 1024 * 1024))
    REQUEST_TIMEOUT = int(os.getenv('REQUEST_TIMEOUT', 30))
    
    # 缓存配置
    CACHE_TYPE = os.getenv('CACHE_TYPE', 'simple')
    CACHE_DEFAULT_TIMEOUT = int(os.getenv('CACHE_DEFAULT_TIMEOUT', 300))
    
    # 上传配置
    UPLOAD_FOLDER = os.getenv('UPLOAD_FOLDER', 'uploads')
    ALLOWED_EXTENSIONS = {'txt', 'pdf', 'png', 'jpg', 'jpeg', 'gif', 'xlsx', 'xls'}


class DevelopmentConfig(Config):
    """开发环境配置"""
    DEBUG = True
    LOG_LEVEL = 'DEBUG'
    
    # 开发环境使用SQLite
    SQLALCHEMY_DATABASE_URI = 'sqlite:///instance/dev.db'


class ProductionConfig(Config):
    """生产环境配置"""
    DEBUG = False
    TESTING = False
    
    # 生产环境必须使用安全的Cookie设置
    SESSION_COOKIE_SECURE = True
    
    # 生产环境日志级别
    LOG_LEVEL = 'WARNING'
    
    # 生产环境限流更严格
    RATELIMIT_DEFAULT = "500 per hour"


class TestingConfig(Config):
    """测试环境配置"""
    TESTING = True
    DEBUG = True
    
    # 测试环境使用内存数据库
    SQLALCHEMY_DATABASE_URI = 'sqlite:///:memory:'
    
    # 禁用限流
    RATELIMIT_ENABLED = False


# 配置映射
config_by_name = {
    'development': DevelopmentConfig,
    'production': ProductionConfig,
    'testing': TestingConfig,
    'default': DevelopmentConfig
}


def get_config(env=None):
    """获取配置类
    
    Args:
        env: 环境名称，默认从FLASK_ENV环境变量读取
    
    Returns:
        Config: 配置类
    """
    if env is None:
        env = os.getenv('FLASK_ENV', 'development')
    
    return config_by_name.get(env, config_by_name['default'])
```

## 配置验证

创建配置验证脚本 `validate_config.py`：

```python
#!/usr/bin/env python3
"""
配置验证脚本

确保所有必需的配置都已正确设置
"""

import os
from dotenv import load_dotenv

load_dotenv()

REQUIRED_CONFIGS = [
    'SECRET_KEY',
    'JWT_SECRET_KEY',
    'MQTT_BROKER',
    'MQTT_PORT',
    'MQTT_USERNAME',
    'MQTT_PASSWORD'
]

OPTIONAL_CONFIGS = [
    'REDIS_HOST',
    'REDIS_PORT',
    'DATABASE_URL',
    'LOG_LEVEL'
]

def validate_config():
    errors = []
    warnings = []
    
    # 检查必需配置
    for config in REQUIRED_CONFIGS:
        value = os.getenv(config)
        if not value:
            errors.append(f"缺少必需配置: {config}")
        elif config == 'SECRET_KEY' and value == 'dev-secret-key':
            warnings.append(f"{config} 使用默认值，请修改为安全的密钥")
    
    # 检查可选配置
    for config in OPTIONAL_CONFIGS:
        value = os.getenv(config)
        if not value:
            warnings.append(f"未设置可选配置: {config} (使用默认值)")
    
    # 验证配置值
    mqtt_port = os.getenv('MQTT_PORT')
    if mqtt_port:
        try:
            port = int(mqtt_port)
            if port < 1 or port > 65535:
                errors.append(f"MQTT端口无效: {port}")
        except ValueError:
            errors.append(f"MQTT端口必须是数字: {mqtt_port}")
    
    # 输出结果
    print("=" * 60)
    print("配置验证结果")
    print("=" * 60)
    
    if warnings:
        print("\n⚠️  警告:")
        for warning in warnings:
            print(f"  - {warning}")
    
    if errors:
        print("\n❌ 错误:")
        for error in errors:
            print(f"  - {error}")
        print("\n请修复上述错误后再启动应用")
        return False
    else:
        print("\n✅ 所有配置验证通过")
        return True

if __name__ == '__main__':
    import sys
    sys.exit(0 if validate_config() else 1)
```

## 最佳实践

### 1. 环境变量命名规范

- 使用大写字母和下划线
- 添加前缀区分不同模块（如 `MQTT_`, `JWT_`, `REDIS_`）
- 使用有意义的名称

### 2. 敏感信息处理

```bash
# ❌ 不好：将敏感信息直接写在代码中
SECRET_KEY = 'my-secret-key'

# ✅ 好：从环境变量读取
SECRET_KEY = os.getenv('SECRET_KEY')
```

### 3. 默认值设置

```python
# 总是提供合理的默认值
TIMEOUT = int(os.getenv('TIMEOUT', 30))  # 默认30秒
DEBUG = os.getenv('DEBUG', 'False').lower() == 'true'  # 默认关闭调试
```

### 4. 配置文档化

在 `.env.example` 中注释每个配置项：

```bash
# Flask密钥（必需）
SECRET_KEY=your-secret-key-here

# MQTT配置
MQTT_BROKER=your-broker.com
MQTT_PORT=8883
```

### 5. 环境隔离

```
.env.development    # 开发环境
.env.production     # 生产环境
.env.testing        # 测试环境
```

## 常见问题

### Q: 如何在不重启应用的情况下更新配置？

A: 对于某些配置，可以使用配置重载功能：

```python
@app.route('/api/admin/reload-config', methods=['POST'])
@requires_admin
def reload_config():
    load_dotenv(override=True)
    return {'success': True, 'message': '配置已重新加载'}
```

### Q: 如何在不同环境使用不同数据库？

A: 在配置类中设置不同的数据库URI：

```python
class ProductionConfig(Config):
    SQLALCHEMY_DATABASE_URI = os.getenv('PROD_DATABASE_URL')
```

### Q: 如何加密敏感配置？

A: 使用配置加密工具或密钥管理服务：

```python
from cryptography.fernet import Fernet

# 解密配置
cipher_suite = Fernet(fernet_key)
decrypted_value = cipher_suite.decrypt(encrypted_config)
```
